const crypto = require('crypto'); function calcolaRisultato(serverSeed, clientSeed, nonce) { // 1. Combiniamo il seed dell'utente con il numero della giocata const message = `${clientSeed}:${nonce}`; // 2. Creiamo un hash HMAC-SHA256 usando il Server Seed come chiave segreta const hash = crypto.createHmac('sha256', serverSeed) .update(message) .digest('hex'); // 3. Prendiamo i primi 8 caratteri dell'hash esadecimale const partialHash = hash.substring(0, 8); // 4. Lo convertiamo in un numero decimale const decimalValue = parseInt(partialHash, 16); // 5. Calcoliamo un risultato da 0.00 a 100.00 (Standard per i dadi) // 4294967295 è il valore massimo per 8 caratteri hex (0xffffffff) const roll = (decimalValue / 4294967295) * 100; return roll.toFixed(2); } // --- SIMULAZIONE --- const serverSeed = 'stringa_segreta_del_server_generata_prima_del_roll'; const clientSeed = 'stringa_casuale_scelta_dal_giocatore'; const nonce = 1; const risultato = calcolaRisultato(serverSeed, clientSeed, nonce); console.log(`Il risultato del lancio è: ${risultato}`);
